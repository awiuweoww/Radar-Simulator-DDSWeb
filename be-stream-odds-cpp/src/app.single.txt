/**
 * Konfigurasi Stress Test
 */
struct SimulationConfig {
    bool squareEnabled   = true;
    bool circleEnabled   = true;
    bool triangleEnabled = true;
    bool parallelogramEnabled   = true;
    bool trapezoidEnabled   = true;
    bool rhombusEnabled   = true;
    bool ellipseEnabled   = true;
    bool pentagonEnabled   = true;
    bool hexagonEnabled   = true;
    int stressObjectCount = 100;
};

/**
 * Generator angka acak sederhana berbasis seed.
 */
double get_random(int index, double seed) {
    return fmod(std::abs(sin(index * 12.9898 + seed * 78.233)) * 43758.5453, 1.0);
}

/**
 * Mengambil timestamp millisecond dari system_clock.
 * Dipanggil SEKALI per burst — seluruh track dalam satu siklus
 * menggunakan timestamp yang sama agar baseline pengukuran latency konsisten.
 */
static inline long long get_burst_timestamp_ms() {
    return std::chrono::duration_cast<std::chrono::milliseconds>(
        std::chrono::system_clock::now().time_since_epoch()
    ).count();
}

int main(int argc, char* argv[]) {
    try {
        LOG_INFO("Main", " [Backend] Mesin radar OpenDDS v1.2 - MODE STREAMING (FIXED) dimulai...");

        ConfigUtil::create_instance();

        /** Inisialisasi Domain Participant */
        DDS::DomainParticipantFactory_var factory = TheParticipantFactoryWithArgs(argc, argv);
        auto participant = ODDSDomainParticipant::get_instance(factory)->get_participant_();

        /** Inisialisasi Publisher */
        RadarTrackODDSPublisher radar_publisher(participant);

        // Inisialisasi Stress Publishers
        SquareTrackODDSPublisher square_publisher(participant);
        CircleTrackODDSPublisher circle_publisher(participant);
        TriangleTrackODDSPublisher triangle_publisher(participant);
        ParallelogramTrackODDSPublisher parallelogram_publisher(participant);
        TrapezoidTrackODDSPublisher trapezoid_publisher(participant);
        RhombusTrackODDSPublisher rhombus_publisher(participant);
        EllipseTrackODDSPublisher ellipse_publisher(participant);
        PentagonTrackODDSPublisher pentagon_publisher(participant);
        HexagonTrackODDSPublisher hexagon_publisher(participant);

        SimulationConfig config;

        /** Inisialisasi Subscriber dan Handler */
        RadarCommandODDSSubscriber radar_subscriber(participant);
        RadarHandler radar_handler(&radar_subscriber);
        radar_subscriber.add_observer(&radar_handler);
        std::thread subscriber_thread(
            BackendInterfaceThreadsContainer::radar_receiver_thread,
            &radar_subscriber
        );

        auto startTime = std::chrono::steady_clock::now();

        /**
         * Pre-kalkulasi parameter statis setiap track.
         * Semua nilai ini deterministik (berbasis index + seed konstan).
         */
        struct TrackParams {
            double startLat;
            double startLon;
            float  speed;
            double sinHeading;
            double cosHeading;
            uint8_t classification;
        };

        const int MAX_TRACKS = 10000;
        std::vector<TrackParams> params(MAX_TRACKS);
        for (int i = 0; i < MAX_TRACKS; i++) {
            double headingRad    = get_random(i, 3.3) * 6.28318;
            double knots         = 100.0 + (get_random(i, 5.5) * 400.0);
            double velocityFactor = knots * 0.0000035;

            params[i].startLat      = -5.5 + (get_random(i, 1.1) - 0.5) * 1.5;
            params[i].startLon      = 110.5 + (get_random(i, 2.2) - 0.5) * 1.5;
            params[i].speed         = (float)knots;
            params[i].sinHeading    = sin(headingRad) * velocityFactor;
            params[i].cosHeading    = cos(headingRad) * velocityFactor;
            params[i].classification = (uint8_t)(get_random(i, 6.6) > 0.6 ? 1 : 0);
        }

        LOG_INFO("Main", "Prakalkulasi selesai. Memulai perulangan simulasi...");

        while (true) {
            int currentCount       = radar_handler.get_target_count();
            int64_t cmdReceivedAt  = radar_handler.get_command_received_at();

            auto nowSteady = std::chrono::steady_clock::now();
            double timeSec = std::chrono::duration_cast<std::chrono::milliseconds>(
                nowSteady - startTime
            ).count() * 0.001;

            // =========================================================
            // FIX: Ambil satu timestamp utama untuk seluruh siklus.
            // Ini memastikan data RADAR dan STRESS (SQUARE, dsb) 
            // memiliki ID siklus yang sama untuk Race Condition Monitor.
            // =========================================================
            const long long commonBurstTimestamp = get_burst_timestamp_ms();

            for (int i = 0; i < currentCount; i++) {
                const auto& p = params[i];
                RadarTrackTransmitData t;
                t.trackId           = i;
                t.lat               = p.startLat + (p.sinHeading * timeSec);
                t.lon               = p.startLon + (p.cosHeading * timeSec);
                t.speed             = p.speed;
                t.timestamp         = commonBurstTimestamp; 
                t.classification    = p.classification;
                t.commandReceivedAt = cmdReceivedAt;

                radar_publisher.set_radar_data(t);
            }

            if (currentCount > 0) {
                LOG_INFO("Main", "Simulation Cycle: Published "
                    + std::to_string(currentCount)
                    + " tracks (ID: 0 - "
                    + std::to_string(currentCount - 1)
                    + ") | T=" + std::to_string(commonBurstTimestamp) + "ms");
            }

            // --- STRESS TEST LOGIC ---
            // 1. Square Stress
            if (config.squareEnabled) {
                for (int i = 0; i < config.stressObjectCount; i++) {
                    StressTrackTransmitData s;
                    s.trackId   = i;
                    s.lat       = -5.0 + (i * 0.0001);
                    s.lon       = 110.0 + (i * 0.0001);
                    s.timestamp = commonBurstTimestamp; 
                    square_publisher.set_square_data(s);
                }
                LOG_INFO("Main", "Stress Test: Published "
                    + std::to_string(config.stressObjectCount) + " Square objects.");
            }

            // 2. Circle Stress
            if (config.circleEnabled) {
                for (int i = 0; i < config.stressObjectCount; i++) {
                    StressTrackTransmitData c;
                    c.trackId   = i;
                    c.lat       = -6.0 + (i * 0.0001);
                    c.lon       = 111.0 + (i * 0.0001);
                    c.timestamp = commonBurstTimestamp;
                    circle_publisher.set_circle_data(c);
                }
                LOG_INFO("Main", "Stress Test: Published "
                    + std::to_string(config.stressObjectCount) + " Circle objects.");
            }

            // 3. Triangle Stress
            if (config.triangleEnabled) {
                for (int i = 0; i < config.stressObjectCount; i++) {
                    StressTrackTransmitData t;
                    t.trackId   = i;
                    t.lat       = -7.0 + (i * 0.0001);
                    t.lon       = 112.0 + (i * 0.0001);
                    t.timestamp = commonBurstTimestamp;
                    triangle_publisher.set_triangle_data(t);
                }
                LOG_INFO("Main", "Stress Test: Published "
                    + std::to_string(config.stressObjectCount) + " Triangle objects.");
            }

            // 4. Parallelogram Stress
            if (config.parallelogramEnabled) {
                for (int i = 0; i < config.stressObjectCount; i++) {
                    StressTrackTransmitData p;
                    p.trackId   = i;
                    p.lat       = -8.0 + (i * 0.0001);
                    p.lon       = 113.0 + (i * 0.0001);
                    p.timestamp = commonBurstTimestamp;
                    parallelogram_publisher.set_parallelogram_data(p);
                }
                LOG_INFO("Main", "Stress Test: Published "
                    + std::to_string(config.stressObjectCount) + " Parallelogram objects.");
            }

            // 5. Trapezoid Stress
            if (config.trapezoidEnabled) {
                for (int i = 0; i < config.stressObjectCount; i++) {
                    StressTrackTransmitData trz;
                    trz.trackId   = i;
                    trz.lat       = -3.0 + (i * 0.0001);
                    trz.lon       = 112.0 + (i * 0.0001);
                    trz.timestamp = commonBurstTimestamp;
                    trapezoid_publisher.set_trapezoid_data(trz);
                }
                LOG_INFO("Main", "Stress Test: Published "
                    + std::to_string(config.stressObjectCount) + " Trapezoid objects.");
            }

            // 6. Rhombus Stress
            if (config.rhombusEnabled) {
                for (int i = 0; i < config.stressObjectCount; i++) {
                    StressTrackTransmitData r;
                    r.trackId   = i;
                    r.lat       = -4.0 + (i * 0.0001);
                    r.lon       = 108.0 + (i * 0.0001);
                    r.timestamp = commonBurstTimestamp;
                    rhombus_publisher.set_rhombus_data(r);
                }
                LOG_INFO("Main", "Stress Test: Published "
                    + std::to_string(config.stressObjectCount) + " Rhombus objects.");
            }

            // 7. Ellipse Stress
            if (config.ellipseEnabled) {
                for (int i = 0; i < config.stressObjectCount; i++) {
                    StressTrackTransmitData el;
                    el.trackId   = i;
                    el.lat       = -7.0 + (i * 0.0001);
                    el.lon       = 108.0 + (i * 0.0001);
                    el.timestamp = commonBurstTimestamp;
                    ellipse_publisher.set_ellipse_data(el);
                }
                LOG_INFO("Main", "Stress Test: Published "
                    + std::to_string(config.stressObjectCount) + " Ellipse objects.");
            }

            // 8. Pentagon Stress
            if (config.pentagonEnabled) {
                for (int i = 0; i < config.stressObjectCount; i++) {
                    StressTrackTransmitData pen;
                    pen.trackId   = i;
                    pen.lat       = -9.0 + (i * 0.0001);
                    pen.lon       = 110.0 + (i * 0.0001);
                    pen.timestamp = commonBurstTimestamp;
                    pentagon_publisher.set_pentagon_data(pen);
                }
                LOG_INFO("Main", "Stress Test: Published "
                    + std::to_string(config.stressObjectCount) + " Pentagon objects.");
            }

            // 9. Hexagon Stress
            if (config.hexagonEnabled) {
                for (int i = 0; i < config.stressObjectCount; i++) {
                    StressTrackTransmitData hex;
                    hex.trackId   = i;
                    hex.lat       = -5.0 + (i * 0.0001);
                    hex.lon       = 114.0 + (i * 0.0001);
                    hex.timestamp = commonBurstTimestamp;
                    hexagon_publisher.set_hexagon_data(hex);
                }
                LOG_INFO("Main", "Stress Test: Published "
                    + std::to_string(config.stressObjectCount) + " Hexagon objects.");
            }

            std::this_thread::sleep_for(std::chrono::milliseconds(1000));
        }

        if (subscriber_thread.joinable()) {
            subscriber_thread.join();
        }
    }
    catch (const std::exception &err) {
        LOG_ERROR("Main", std::string("Kesalahan Fatal (std): ") + err.what());
    }
    catch (...) {
        LOG_ERROR("Main", "Kesalahan Fatal: Terjadi eksepsi tidak dikenal (kemungkinan eksepsi CORBA/DDS)");
    }

    return 0;
}

Kode resource_limits adalah cara kita menentukan kapasitas gudang penyimpanan di dalam memori DDS.

Saat Anda menggunakan RELIABLE (pasti sampai) dan KEEP_ALL (simpan semua), DDS berjanji tidak akan membuang data. Tapi, janji itu dibatasi oleh berapa besar memori yang kita siapkan.

Berikut penjelasan masing-masing barisnya:

max_samples = 5000: Ini adalah kapasitas total antrian (antrian global). Karena kita punya 4 topik yang masing-masing mengirim 100 data secara bersamaan (total 400 data/detik), kita butuh "gudang" yang cukup besar untuk menampung data tersebut jika seandainya Gateway atau Browser sedang sibuk memproses hal lain.
max_instances = 100: Ini adalah jumlah objek unik yang bisa dikenali (berdasarkan trackId). Jika simulasi Anda mengirim lebih dari 100 ID unik, angka ini perlu dinaikkan. (Untuk pengujian stress saat ini, 100 biasanya cukup).
max_samples_per_instance = 1000: Ini adalah berapa banyak histori data yang disimpan per satu objek. Misalnya, jika data untuk trackId 5 datang berkali-kali sangat cepat sebelum sempat diproses, DDS bisa menampung hingga 1000 histori untuk ID tersebut.
Intinya: Tanpa resource_limits yang besar, saat Anda melakukan "Real Race" (semua thread menembak data serentak), antrian DDS yang aslinya kecil (default biasanya hanya 100-200) akan langsung penuh. Begitu penuh, DDS akan membuang data baru, itulah yang menyebabkan status MISSING pada laporan Anda sebelumnya.

Dengan menyetel ke 5000, kita memberikan ruang yang sangat luas agar tidak ada data yang terbuang saat lonjakan trafik terjadi.

// Resource Limits for RELIABLE + KEEP_ALL
    this->w_qos_.resource_limits.max_samples = 5000;
    this->w_qos_.resource_limits.max_instances = 100;
    this->w_qos_.resource_limits.max_samples_per_instance = 1000;

//////

Jawaban untuk pertanyaan "ini untuk apa yaa?":

Blok kode di atas adalah konfigurasi QoS (Quality of Service) untuk membatasi kapasitas penyimpanan internal (resource_limits) yang dialokasikan oleh DDS pada DataWriter (atau DataReader) ketika menggunakan keandalan RELIABLE dan sejarah KEEP_ALL.

Mengapa konfigurasi ini diperlukan?
1. Menghindari Pemblokiran / Data Hilang: Secara default, kapasitas antrean internal DDS sangat terbatas. Saat program mengirimkan ribuan objek stress test sekaligus secara cepat, antrean DDS akan cepat penuh. Ketika penuh, DDS dengan QoS RELIABLE + KEEP_ALL akan memblokir thread pengirim (DataWriter) atau membuang data baru, yang menyebabkan status "MISSING" pada penerimaan data.
2. Penjelasan Batasan yang Diatur:
   - max_samples = 5000: Batas total sampel data maksimum yang dapat disimpan di memori publisher secara keseluruhan sebelum dikirim dan mendapat konfirmasi (ACK) dari subscriber.
   - max_instances = 100: Batas jumlah objek/track unik (berdasarkan field kunci seperti trackId) yang dapat dikelola secara aktif oleh DataWriter dalam satu waktu.
   - max_samples_per_instance = 1000: Batas riwayat sampel data yang disimpan untuk setiap track/instance tunggal.

Dengan mengatur batas-batasan ini menjadi lebih tinggi (seperti 5000), program memberikan ruang buffer memori yang sangat luas kepada DDS untuk mengatasi lonjakan trafik (burst) selama pengujian beban berat tanpa kehilangan data.
